## 复核命令

在仓库根目录运行：

```sh
xcodebuild -project ios/Maomao.xcodeproj -scheme Maomao \
  -configuration Debug -destination 'generic/platform=iOS Simulator' \
  -derivedDataPath /tmp/cattt-check-build CODE_SIGNING_ALLOWED=NO build

# 资源检查需要 Python 3 和 Pillow。
python3 assets/tools/validate_world_v4.py \
  /tmp/cattt-check-build/Build/Products/Debug-iphonesimulator/Maomao.app

# 独立逻辑测试，无须在 Xcode 中创建测试 target。
check_dir=$(mktemp -d /tmp/cattt-tests.XXXXXX)
for test in ios/Tests/*.swift; do
  executable="$check_dir/$(basename "$test" .swift)"
  swiftc ios/Maomao/Model/WorldScene.swift \
    ios/Maomao/Model/WorldInteraction.swift \
    ios/Maomao/Model/WorldCamera.swift "$test" -o "$executable" || exit 1
  "$executable" assets/scenes/courtyard-v4/scene.json || exit 1
done
```
